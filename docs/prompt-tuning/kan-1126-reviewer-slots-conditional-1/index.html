+// pstChangedLines parses a -U0 diff into each path's changed lines. A
+// `diff --git` line opens a file header, which ends at its first `@@`; in
+// it `--- a/<p>` and `+++ b/<p>` name the path (`---` for a deletion, whose
+// `+++` is /dev/null). Each hunk header sets the old and new counters; a
+// `+` line is added at the new one, a `-` line removed at the old one.
+func pstChangedLines(diff string) map[string][]pstLine {
+	out := map[string][]pstLine{}
+	var p string
+	var header bool
+	var oldN, newN int
+	for _, l := range strings.Split(diff, "\n") {
+		switch {
+		case strings.HasPrefix(l, "diff --git "):
+			header, p = true, ""
+		case header && strings.HasPrefix(l, "--- a/"):
+			p = strings.TrimPrefix(l, "--- a/")
+		case header && strings.HasPrefix(l, "+++ b/"):
+			p = strings.TrimPrefix(l, "+++ b/")
+		case strings.HasPrefix(l, "@@"):
+			header = false
+			if m := pstHunk.FindStringSubmatch(l); m != nil {
+				oldN, _ = strconv.Atoi(m[1])
+				newN, _ = strconv.Atoi(m[2])
+			}
+		case header:
+		case strings.HasPrefix(l, "+"):
+			out[p] = append(out[p], pstLine{newN, l[1:]})
+			newN++
+		case strings.HasPrefix(l, "-"):
+			out[p] = append(out[p], pstLine{oldN, l[1:]})
+			oldN++
+		}
+	}
+	return out
+}

Why picked

1. Defect class: code reads a path or name out of a command's human-oriented output (a diff header, a status or name-only line) by trimming a fixed prefix, while the tool quotes or escapes names holding a space, a quote, a backslash, a control character or a non-ASCII byte, so those names are silently mis-keyed or dropped. 2. No current rule covers it: engineering-principles.md's Robustness section (quoted as `before`) says to be defensive about what you accept, but its violations list names catch blocks, timeouts, fallbacks and schema invariants and nothing about consuming another tool's output; no paragraph of implementer-brief.md, implement.md, sdd-dispatch.md or engineering-principles.md mentions reading a tool's machine-readable form or its quoting. 3. It generalises: the same class was fixed earlier in this repository's history at least twice (a script classifying dirty paths that had to read them literally, unquoted; a plan-shape guard that had to add core.quotePath=false), it recurred here as an Important finding, and the round's own fix handled only the two quoted forms the finding named, leaving names with a quote or a backslash still dropped; it applies to any code parsing git, ls, ps, docker or any other CLI's text output, in any language. 4. It is a principle, not a patch: "read a tool's machine form or decode its whole quoting grammar" names no tool flag, file or guard.
