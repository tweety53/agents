+func numberRefLessFindings(ctx context.Context, path string, data []byte, rest []string, stderr io.Writer) ([]byte, int) {
+	lines := strings.Split(string(data), "\n")
+	var refLess []int
+	rows := map[int]map[string]any{}
+	floor := 1
+	for i, line := range lines {
+		if strings.TrimSpace(line) == "" {
+			continue
+		}

+		dec := json.NewDecoder(strings.NewReader(line))
+		dec.UseNumber()

+			continue
+		}
+		if ref, _ := row["ref"].(string); ref != "" {
+			if n, err := strconv.Atoi(strings.TrimPrefix(ref, "F")); err == nil {
+				floor = max(floor, n+1)
+			}
+			continue
+		}

+// replaceFile writes data to a temp file beside path and renames it over
+// path, keeping path's mode, so a failed write leaves the original intact
+// and no temp file behind.
+func replaceFile(path string, data []byte) error {
+	fi, err := os.Stat(path)
+	if err != nil {
+		return err
+	}
+	tmp, err := os.CreateTemp(filepath.Dir(path), filepath.Base(path)+".tmp-*")
+	if err != nil {
+		return err
+	}
+	defer os.Remove(tmp.Name())
+	if _, err := tmp.Write(data); err != nil {
+		_ = tmp.Close()
+		return err
+	}
+	if err := tmp.Close(); err != nil {
+		return err
+	}
+	if err := os.Chmod(tmp.Name(), fi.Mode().Perm()); err != nil {
+		return err
+	}
+	return os.Rename(tmp.Name(), path)

Why picked

1. Defect class: an in-place truncating rewrite of a sole-copy file, where an interrupted write destroys the data the rewrite meant to update. 2. No rule covers it: the robustness principle's violation list reads "an empty `catch` block, a swallowed exception logged at debug, an external call with no timeout or bounded retry, a fallback that silently returns empty data, an error message that omits the identifier needed to investigate it, or an invariant the schema could enforce …" and names nothing about how a file is replaced; the implementer wrote `os.WriteFile(path, out, 0o644)` over the findings file it had just read, and the failure-modes slot raised it as an Important that cost a fix chunk. 3. It generalises: kan-1032's panel raised the same shape on `stats/cmd/flow/prompttuning.go:36` ("os.WriteFile is not atomic; a truncated index.html is possible"), and the same holds for any state file, config file, journal or lockfile a CLI or service rewrites, in every language — the repository's own renderers already write atomically (render-slot-prompt.sh). 4. It is a principle, not a patch: replace a file whose old bytes are the only copy by writing a temp file beside it and renaming it over, never by truncating it.
