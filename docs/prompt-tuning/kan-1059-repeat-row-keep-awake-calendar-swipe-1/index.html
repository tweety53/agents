+async function recordWakeLocks(page: Page): Promise<void> {
+  await page.addInitScript(() => {
+    const w = window as any;
+    w.__locks = 0;
+    w.__lockRequests = 0;
+    Object.defineProperty(navigator, 'wakeLock', {
+      configurable: true,
+      value: {
+        request: async () => {
+          w.__lockRequests++;
+          w.__locks++;
+          let released = false;
+          const s: any = new EventTarget();
+          s.type = 'screen';
+          Object.defineProperty(s, 'released', { get: () => released });
+          s.release = async () => {
+            if (!released) {
+              released = true;
+              w.__locks--;
+              s.dispatchEvent(new Event('release'));
+            }
+          };
+          return s;
+        },
+      },
+    });
+  });
+}
+
+const locks = (page: Page) => page.evaluate(() => (window as any).__locks as number);

+async function park(page: Page): Promise<void> {
+  await page.mouse.move(0, 0);

+async function drag(page: Page, dx: number, tag = 'home_calendar_swipe'): Promise<void> {
+  const box = (await tagBox(page, tag))!;
+  const y = box.y + box.height / 2;
+  const x0 = box.x + box.width / 2;

+  const steps = 12;
+  for (let i = 1; i <= steps; i++) await page.mouse.move(x0 + (dx * i) / steps, y);
+  await page.mouse.up();

Why picked

1. Defect class: a probe whose input conditions differ from those the defect was observed under passes over the defect. 2. The quoted rule fixes what the probe asserts (the measured value, the real component) but not the input it drives or when it samples; no rule in implement.md, sdd-dispatch.md, engineering-principles.md or verify-fix-loop.md covers input fidelity. 3. It generalises to every motion, gesture, debounce, animation and race defect, and to backend races probed single-threaded; gymie's own frontend rules record the class twice (KAN-575 "A green suite is not evidence of a frame"; the Compose test clock that cannot see the frame before a LaunchedEffect runs). 4. It states a principle (reproduce the observed conditions, sample inside the observed window), not a special case for calendars or presses.
